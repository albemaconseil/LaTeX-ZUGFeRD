\ExplSyntaxOff

% French number format (decimal comma) for all amounts.
% (zugferd-invoice only sets the locale for German documents.)
\sisetup{locale=FR}
\renewcommand*{\invoicesumnettoname}{Total HT}
\renewcommand*{\invoicesumbruttoname}{Total TTC}
% The demo implementation only knows the VAT rates 19 and 7 by default.
\InitVAT{20}

% Count the elements of a comma separated list (number of days worked)
\ExplSyntaxOn
\NewExpandableDocumentCommand{\CountList}{m}{\clist_count:N #1}
\ExplSyntaxOff

\def\FactureMonth        {avril}
\def\FactureFromMonth    {d'avril}
\def\FactureMonthLastDay {30}
\def\FactureMonthNumber  {04}
\def\FactureYearNumber   {26}
\def\FactureYear         {20\FactureYearNumber}
\def\FactureDaysDetails  {1, 2, 7, 8, 9, 13, 14, 15, 16, 20, 21, 22, 27, 28, 29}
\edef\FactureDays{\CountList{\FactureDaysDetails}}% number of days = length of the list
\def\FactureAmountPerDay {480}

\def\FactureNum      {\FactureYearNumber-00\FactureMonthNumber}% invoice number
\def\FactureDate     {\FactureMonthLastDay~\FactureMonth~\FactureYear}% printed date
\def\FactureDateYMD  {\FactureYear\FactureMonthNumber\FactureMonthLastDay}% date as YYYYMMDD

\SetZUGFeRDData{
%	document-type = commercial-invoice, % commented as this setting matches the initial value
	id=\FactureNum,
	date=\FactureDateYMD,
	delivery-date=\FactureDateYMD,
	unit=day,
	currency=EUR,
	payment-terms={Payable dans les 30 jours},% either this one or the due-date is required
	% due-date={20260530},
	start-date=\FactureYear\FactureMonthNumber01,% invoicing period: first to last day of the month
	end-date=\FactureDateYMD,
	business-process=S1,% invoicing framework: services. Check the value with your platform.
}

% Do not use ~ in the description: this text is also written to the XML as it is.
\AddInvoiceItem<unit=DAY>{\FactureDays}{Développements \FactureMonth\space\FactureYear}{\FactureAmountPerDay}
