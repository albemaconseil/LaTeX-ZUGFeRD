% --- Legal notices (articles L441-10 and D441-5 of the French Code de commerce)
% No ~ or special symbols: the texts are also written to the XML (notes AAB, PMD, PMT)
\newcommand*{\MentionEscompte}{Pas d'escompte pour paiement anticipé.}
\newcommand*{\MentionPenalites}{Pénalités de retard : 3 fois le taux d'intérêt légal.}
\newcommand*{\MentionIndemnite}{Indemnité forfaitaire pour frais de recouvrement en cas de retard de paiement : 40 euros.}

% --- Helpers: split the IBAN, format the SIREN
\ExplSyntaxOn
\NewExpandableDocumentCommand{\StrRange}{mmm}{\exp_args:Nf\str_range:nnn{#1}{#2}{#3}}
\ExplSyntaxOff
\newcommand*{\IBANspaced}[1]{%
	\StrRange{#1}{1}{4} \StrRange{#1}{5}{8} \StrRange{#1}{9}{12} \StrRange{#1}{13}{16} %
	\StrRange{#1}{17}{20} \StrRange{#1}{21}{24} \StrRange{#1}{25}{27}}
\newcommand*{\RIBBanque}{\StrRange{\BankIBAN}{5}{9}}
\newcommand*{\RIBGuichet}{\StrRange{\BankIBAN}{10}{14}}
\newcommand*{\RIBCompte}{\StrRange{\BankIBAN}{15}{25}}
\newcommand*{\RIBCle}{\StrRange{\BankIBAN}{26}{27}}
% ordinary spaces (no ~): also used in the XML
\def\SirenFmt#1#2#3#4#5#6#7#8#9{#1#2#3 #4#5#6 #7#8#9}
\newcommand*{\SellerSirenSpaced}{\expandafter\SirenFmt\SellerSiren}

% --- Seller and payment data for the Factur-X XML
\SetZUGFeRDData{
	seller/name = {\SellerName},
	seller/postcode = {\SellerPostcode},
	seller/city = {\SellerCity},
	seller/country = FR,
	seller/address = {\SellerStreet},
	seller/vatid = {\SellerVAT},
	seller/legal-id = {\SellerSiren},% SIREN (schemeID 0002, set by format=facturx)
	seller/contact = {\SellerContactName\\\SellerMobile\\\SellerEmail},
	seller/email = {\SellerEmail},
	note-AAB = {\MentionEscompte},% discount conditions
	note-PMD = {\MentionPenalites},% late payment penalties
	note-PMT = {\MentionIndemnite},% fixed recovery fee
	payment-means / type = 58, % SEPA credit transfer
	payment-means / iban = {\BankIBAN},% without spaces in the XML
	payment-means / account-holder = {\SellerName},
	payment-means / bic = {\BankBIC},
}

% --- Page footer
\cfoot{\small
	\SellerName, SARL au capital de \SellerCapital{} € ~--~ \SellerStreet{} - \SellerPostcode{} \SellerCity\newline
	Téléphone : \SellerPhone{} ~--~ Site web : \SellerWeb{} ~--~ E-mail : \SellerContactEmail\newline
	RCS \SellerRCSCity{} \SellerSirenSpaced{} ~--~ NAF \SellerNAF{} ~--~ TVA intracommunautaire : \SellerVAT\par
	\makebox[\linewidth][r]{\tiny Facture électronique Factur-X (profil EN 16931) générée avec La\TeX.}
}
